Что было до ES6

До появления стандарта ES6, разработчики создавали «псевдомодули». Они использовали стандартные конструкции языка, чтобы реализовать возможности настоящих модулей.

Вот несколько проблем, с которыми сталкивались разработчики до ES6. Обратите внимание, что речь идёт только о модульности до ES6, поэтому в примерах есть let и const.

Редекларирование

Рассмотрим два файла со скриптами. Каждый создаёт DOM-элемент с тегом div. Строки 'div' хранятся в константах:

  • file1.js

    'use strict';


const TAG = 'div';


const element = document.createElement(TAG);


element.textContent = 'Мой первый тег';
document.body.appendChild(element);

  • file2.js

    'use strict';


const TAG = 'div';


const element = document.createElement(TAG);


element.textContent = 'Мой второй тег';
document.body.appendChild(element);

Оба файла подключаются к index.html:

...
<script src="file1.js"></script>
<script src="file2.js"></script>

Но вместо двух надписей на странице, разработчика ждёт только сообщение об ошибке:

Дело в том, что скрипты содержат идентичные названия переменных, да ещё и с ключевым словом const. При подключении второго файла, эти переменные окажутся в общей области видимости. Браузер попытается исполнить второй файл и натолкнётся на повторное объявление const TAG. Интерпретатор такого не позволит и выбросит исключение.

Этот пример показывает, насколько внимательным должен быть разработчик при использовании таких псевдомодулей. Когда файлов становится не два, а десять или сто, контролировать все имена переменных уже невозможно.

Решение проблемы — замыкание.

Под замыканием понимаем лексическое окружение функции и принципы работы языка с этим лексическим окружением. Для создания таких областей разработчики используют паттерн IIFE — самовызывающиеся функции. Любые переменные внутри IIFE невидимы для внешнего мира:

let a = 1;
var b = 2;

const result = (function() {
  let a = 3;
  console.log(a, b); // 3 2
  return a;
})();

console.log(a, b, result); // 1 2 3

Такой паттерн приближает псевдомодули к настоящим.

Вернёмся к исходному примеру, чтобы увидеть, как IIFE поможет решить проблему:

  • file1.js

    'use strict';


(function () {
    const TAG = 'div';


    const element = document.createElement(TAG);


    element.textContent = 'Мой первый тег';
    document.body.appendChild(element);
})();

  • file2.js

    'use strict';


(function () {
    const TAG = 'div';


    const element = document.createElement(TAG);


    element.textContent = 'Мой второй тег';
    document.body.appendChild(element);
})();

  • index.html

    ...
<script src="file1.js"></script>
<script src="file2.js"></script>

Глобальная область видимости

IIFE помогает изолировать переменные, но не всегда модули должны быть полностью отделены от основного кода. Часто разработчики хотят вызывать функции модуля в других блоках кода.

Так можно извне получить доступ к содержимому IIFE:

  • file1.js

    window.utils = (function () {
	// Здесь суперреализации функций, которые хочется переиспользовать в проекте


	function mySuperFunc1() {
		console.log('utils.mySuperFunc1');
	}


	function mySuperFunc2() {
		console.log('utils.mySuperFunc2');
	}


	return {
		mySuperFunc1,
		mySuperFunc2,
	};
})();

  • file2.js

    (function () {
	// Здесь какая-то суперреализация функций, которые хочется переиспользовать много где в проекте


	function myOtherSuperFunc1() {
			window.utils.mySuperFunc1(); // Вызывали из file1.js
			console.log('otherUtils.myOtherSuperFunc1');
	}


	function myOtherSuperFunc2() {
	  console.log('otherUtils.myOtherSuperFunc2');
	}


	myOtherSuperFunc1();


	window.otherUtils = {
	  myOtherSuperFunc1,
	  myOtherSuperFunc2,
	};
})();

  • index.html

    <script src="file1.js"></script>
<script src="file2.js">
	// Сначала покажутся логи при вызове myOtherSuperFunc1 из file2.js:
	// utils.mySuperFunc1
  // otherUtils.myOtherSuperFunc1
</script>


<script>
	// Из файла file1.js 
	window.utils.mySuperFunc1(); // utils.mySuperFunc1
	window.utils.mySuperFunc2(); // utils.mySuperFunc2


	// Из файла file2.js 
	window.otherUtils.myOtherSuperFunc1(); // utils.mySuperFunc1
																				 // otherUtils.myOtherSuperFunc1
	window.otherUtils.myOtherSuperFunc2(); // otherUtils.myOtherSuperFunc2
</script>

Порядок подключений

Ваш псевдомодуль умеет изолировать переменные и открывать внешний доступ к своему функционалу. Но у него всё ещё есть недостатки.

Поменяем порядок подключения файлов в index.html:

<!-- Теперь file2.js сверху -->
<script src="file2.js"></script>
<script src="file1.js"></script> 

<script>
	// Из файла file1.js 
	window.utils.mySuperFunc1(); // utils.mySuperFunc1
	window.utils.mySuperFunc2(); // utils.mySuperFunc2

	// Из файла file2.js 
	window.otherUtils.myOtherSuperFunc1(); // utils.mySuperFunc1
																				 // otherUtils.myOtherSuperFunc1
	window.otherUtils.myOtherSuperFunc2(); // otherUtils.myOtherSuperFunc2
</script>

Часть логики сломается, консоль сообщит о двух ошибках:

Первая ошибка: в файле file2.js была вызвана функция из файла file1.js. Но этот файл ещё не подключён, — по умолчанию браузер подключает скрипты последовательно.

// file2.js

(function () {
	function myOtherSuperFunc1() {
      window.utils.mySuperFunc1(); // Вызывали из file1.js
      console.log('otherUtils.myOtherSuperFunc1');
	}

	// ...

	myOtherSuperFunc1();

	// ...

})();

Если убрать из IIFE вызов функции myOtherSuperFunc1 и сделать его в теге <script>, ошибки не будет:

  • file2.js

    (function () {
	function myOtherSuperFunc1() {
	    window.utils.mySuperFunc1(); // Вызывали из file1.js
	    console.log('otherUtils.myOtherSuperFunc1');
	}


	// ...


	// myOtherSuperFunc1(); —> ошибка исчезнет, и всё будет работать


	// ...


})();

  • index.html

    <!-- Теперь file2.js сверху -->
<script src="file2.js"></script> 
<script src="file1.js"></script>


<script>
	// Из файла file1.js 
	window.utils.mySuperFunc1(); // utils.mySuperFunc1
	window.utils.mySuperFunc2(); // utils.mySuperFunc2


	// Из файла file2.js 
	window.otherUtils.myOtherSuperFunc1(); // utils.mySuperFunc1
																				 // otherUtils.myOtherSuperFunc1
	window.otherUtils.myOtherSuperFunc2(); // otherUtils.myOtherSuperFunc2
</script>

Резюме

До появления ES6 JavaScript позволял создавать полезные псевдомодули. Однако такая реализация сопровождалась неудобствами и рисками ошибок. В том числе нужно было соблюдать порядок подключения.
